# Pre-launch: keep the whole site out of search engines. Every page also
# carries <meta name="robots" content="noindex">.
#
# At launch, flip both: replace "Disallow: /" below with "Allow: /", and
# delete the robots meta tag from index.html, download.html and
# benchmarks.html (404.html keeps its own noindex; it is not a page to
# index). A crawler that obeys "Disallow" never fetches a page, so it never
# sees the meta tag: the two cover each other while both are in place.
User-agent: *
Disallow: /
